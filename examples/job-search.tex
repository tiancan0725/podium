\begin{frame}{The McCall model}
\begin{itemize}
\item Each period an unemployed worker draws an offer $w \sim F$ on $[0,\bar w]$.
\pause
\item Accept: earn $w$ forever. Reject: collect $c$ and search again.
\pause
\item Value of an offer:
\[
V(w) = \max\Big\{ \underbrace{\frac{w}{1-\beta}}_{\text{accept}},\;
\underbrace{c + \beta \int_0^{\bar w} V(w')\,dF(w')}_{\text{reject}} \Big\}
\]
\end{itemize}
\end{frame}

\column{0.5\textwidth}
Accept iff $w \ge w^*$, where
\[
w^* - c = \frac{\beta}{1-\beta}\int_{w^*}^{\bar w} (w'-w^*)\,dF(w').
\]
\pause
\alert{Comparative statics:} a better outside option $c$ or more patience $\beta$ raises $w^*$; a mean-preserving spread in $F$ raises it too.
